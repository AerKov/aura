// Alerts today: a read-only list of today's alerts, newest first.
export default function activate(ctx) {
  ctx.ui.registerPanel({
    title: "Alerts today",
    render(el) {
      const draw = async () => {
        const alerts = await ctx.alerts.recent();
        const today = new Date().toDateString();
        const items = alerts
          .filter((a) => new Date(a.lastAt).toDateString() === today)
          .sort((a, b) => String(b.lastAt).localeCompare(String(a.lastAt)));
        el.replaceChildren();
        const title = document.createElement("h1");
        title.textContent = "Alerts today";
        el.append(title);
        if (items.length === 0) {
          const empty = document.createElement("p");
          empty.className = "empty";
          empty.textContent = "No alerts today.";
          el.append(empty);
          return;
        }
        const list = document.createElement("ul");
        for (const a of items) {
          const li = document.createElement("li");
          const badge = document.createElement("span");
          badge.className = `badge ${a.severity}`;
          badge.textContent = a.severity;
          const name = document.createElement("strong");
          name.textContent = a.title;
          const meta = document.createElement("div");
          meta.className = "meta";
          const time = new Date(a.lastAt).toLocaleTimeString([], {
            hour: "2-digit",
            minute: "2-digit",
          });
          meta.textContent = `${a.source} · ${time}${a.count > 1 ? ` · ×${a.count}` : ""}`;
          const summary = document.createElement("p");
          summary.textContent = a.summary;
          li.append(badge, name, meta, summary);
          list.append(li);
        }
        el.append(list);
      };
      draw();
      setInterval(draw, 30_000);
    },
  });
}
