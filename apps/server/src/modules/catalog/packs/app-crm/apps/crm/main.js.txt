export default function activate(ctx) {
  const STATUS = ["lead", "active", "past"];
  const STAGES = ["lead", "proposal", "won", "lost"];
  ctx.ui.registerPanel({
    title: "CRM",
    render(el) {
      ctx.kit.tabs(el, [
        {
          label: "Clients",
          render(tab) {
            ctx.kit.form(tab, { collection: "clients", title: "New client", fields: ["name", "company", "email", "phone", "status"], options: { status: STATUS }, defaults: { status: "lead" } });
            ctx.kit.table(tab, { collection: "clients", columns: [{ field: "name" }, { field: "company" }, { field: "email" }, { field: "status" }], sort: "name", options: { status: STATUS } });
          },
        },
        {
          label: "Deals",
          render(tab) {
            ctx.kit.board(tab, { collection: "deals", groupBy: "stage", groups: STAGES, title: "title", subtitle: "client" });
            ctx.kit.form(tab, { collection: "deals", title: "New deal", fields: ["title", "client", "amount", "stage", "due"], options: { stage: STAGES }, defaults: { stage: "lead" } });
          },
        },
        {
          label: "Overview",
          render(tab) {
            const row = ctx.kit.row(tab);
            ctx.kit.stat(row, { collection: "clients", label: "Clients" });
            ctx.kit.stat(row, { collection: "deals", label: "Open deals", query: { where: { stage: "proposal" } } });
            ctx.kit.stat(row, { collection: "deals", label: "Won", query: { where: { stage: "won" } } });
            ctx.kit.stat(row, { collection: "deals", label: "Won, total amount", query: { where: { stage: "won" } }, sum: "amount" });
            ctx.kit.chart(tab, { collection: "deals", groupBy: "stage", title: "Deals by stage" });
          },
        },
      ]);
    },
  });
}
